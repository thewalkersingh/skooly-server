-- Step 1 — find the user and identity ids
SELECT u.id as user_id, u.identity_id, ui.phone, ui.email
FROM users u
JOIN user_identity ui ON ui.id = u.identity_id
WHERE ui.phone = '9876543210';
-- Step 2 — delete refresh tokens (FK → users)
DELETE FROM refresh_tokens WHERE user_id = ?;
-- Step 3 — delete OTP records (FK → users)
DELETE FROM otp_records WHERE user_id = ?;
-- Step 4 — delete notifications (no FK, just userId column)
DELETE FROM notifications WHERE user_id = ?;
-- Step 5 — delete user
DELETE FROM users WHERE id = ?;
-- Step 6 — delete identity (now safe — no more FKs pointing to it)
DELETE FROM user_identity WHERE id = ?;


-- Quicker approach — single script
Replace ? with actual IDs:
SET @phone = '9876543210';

SET @identity_id = (SELECT identity_id FROM users u JOIN user_identity ui ON ui.id = u.identity_id WHERE ui.phone = @phone);
SET @user_id = (SELECT id FROM users WHERE identity_id = @identity_id);

DELETE FROM refresh_tokens WHERE user_id = @user_id;
DELETE FROM otp_records WHERE user_id = @user_id;
DELETE FROM notifications WHERE user_id = @user_id;
DELETE FROM users WHERE id = @user_id;
DELETE FROM user_identity WHERE id = @identity_id;


-- Even quicker — disable FK checks temporarily
SET FOREIGN_KEY_CHECKS = 0;
DELETE FROM refresh_tokens WHERE user_id = (SELECT id FROM users WHERE identity_id = (SELECT id FROM user_identity WHERE phone = '9876543210'));
DELETE FROM otp_records WHERE user_id = (SELECT id FROM users WHERE identity_id = (SELECT id FROM user_identity WHERE phone = '9876543210'));
DELETE FROM users WHERE identity_id = (SELECT id FROM user_identity WHERE phone = '9876543210');
DELETE FROM user_identity WHERE phone = '9876543210';
SET FOREIGN_KEY_CHECKS = 1;
